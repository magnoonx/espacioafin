	\begin{frame}
		\frametitle{Ejercicio 1: Vector Tangente Unitario}
		Dada la curva en el espacio $\textbf{r}(t) = (t, t^2, t^3)$, encontrar el vector tangente unitario en el punto $t = 2$.
		\pause
		\begin{itemize}
			\item Vector Tangente Unitario: $\textbf{T}(t) = \frac{\textbf{r}'(t)}{\lvert \textbf{r}'(t) \rvert}$
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Ejercicio 2: Recta Tangente}
		Dada la curva en el espacio $\textbf{r}(t) = (\cos(t), \sin(t), t)$, encontrar la ecuación de la recta tangente en el punto $t = \frac{\pi}{4}$.
		\pause
		\begin{itemize}
			\item Recta Tangente: $\textbf{r}(t) = \textbf{r}(t_0) + (\textbf{r}'(t_0))(t - t_0)$
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Ejercicio 3: Plano Normal}
		Dada la curva en el espacio $\textbf{r}(t) = (e^t, e^{-t}, t)$, encontrar la ecuación del plano normal en el punto $t = 0$.
		\pause
		\begin{itemize}
			\item Plano Normal: $\textbf{n} \cdot (\textbf{r} - \textbf{r}_0) = 0$
		\end{itemize}
	\end{frame}
